600  void insert(const KeyType& new_key, const ValueType& new_value) {
601  auto predecessors = find_predecessors(new_key);
602  if (predecessors[0]->forward_links[0] != terminator &&
603  predecessors[0]->forward_links[0]->node_key == new_key) {
604  predecessors[0]->forward_links[0]->node_value = new_value;
605  return;
606  }
607  size_t new_level = generate_random_level();
608  if (new_level > current_height) {
609  predecessors.resize(new_level, header);
610  current_height = new_level;
611  }
612  SkipNode* new_node = new SkipNode(new_key, new_value, predecessors[0], new_level);
613  for (size_t lvl = 0; lvl < new_level; ++lvl) {
614  new_node->forward_links[lvl] = predecessors[lvl]->forward_links[lvl];
615  predecessors[lvl]->forward_links[lvl] = new_node;
616  }
617  new_node->back_link = predecessors[0];
618  if (new_node->forward_links[0] != terminator) {
619  new_node->forward_links[0]->back_link = new_node;
620  } else {
621  terminator->back_link = new_node;
622  }
623  item_count++;
624  }

631  void erase(const KeyType& key) {
632  auto predecessors = find_predecessors(key);
633  SkipNode* target = predecessors[0]->forward_links[0];
634  if (target == terminator || target->node_key != key) return;
635  for (size_t lvl = 0; lvl < target->forward_links.size(); ++lvl) {
636  if (predecessors[lvl]->forward_links[lvl] == target) {
637  predecessors[lvl]->forward_links[lvl] = target->forward_links[lvl];
638  }
639  }
640  if (target->forward_links[0] != terminator) {
641  target->forward_links[0]->back_link = target->back_link;
642  } else {
643  terminator->back_link = target->back_link;
644  }
645  delete target;
646  item_count--;
647  while (current_height > 1 && header->forward_links[current_height-1] == terminator) {
648  current_height--;
649  }
650  }
